import rclpy
from rclpy.node import Node
from std_msgs.msg import String

class Talker(Node):
    def __init__(self):
        super().__init__('talker')
        self.pub = self.create_publisher(
            String, 'chatter', 10)
        self.create_timer(0.5, self.tick)

    def tick(self):
        msg = String(data='Hello ROS 2')
        self.pub.publish(msg)
